cmake_minimum_required(VERSION 3.20)
project(wendy_nimble_registration C)
include("${CMAKE_CURRENT_LIST_DIR}/../../dependency_patches.cmake")
set(ENV{IDF_PATH} "${WENDY_TEST_IDF_PATH}")
set(CONFIG_BT_NIMBLE_ENABLED ON)
set(host "${WENDY_TEST_IDF_PATH}/components/bt/host/nimble/nimble/nimble/host")
set(names ble_hs.c ble_hs_hci.c ble_l2cap_coc.c ble_l2cap_sig.c)
set(sources)
foreach(name IN LISTS names)
    list(APPEND sources "${host}/src/${name}")
    file(SHA256 "${host}/src/${name}" original_${name})
endforeach()
add_library(__idf_bt STATIC ${sources})
target_include_directories(__idf_bt PUBLIC "${host}/include" PRIVATE "${host}/src")
foreach(pass RANGE 1 2)
    wendy_patch_nimble()
    get_target_property(actual __idf_bt SOURCES)
    foreach(name IN LISTS names)
        set(expected "${CMAKE_BINARY_DIR}/wendy_patched/nimble_host/src/${name}")
        if(NOT expected IN_LIST actual)
            message(FATAL_ERROR "NimBLE target retained unpatched ${name}: ${actual}")
        endif()
        file(SHA256 "${host}/src/${name}" unchanged)
        if(NOT unchanged STREQUAL original_${name})
            message(FATAL_ERROR "Installed SDK changed: ${name}")
        endif()
    endforeach()
    foreach(property INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
        get_target_property(actual __idf_bt ${property})
        if(NOT "${CMAKE_BINARY_DIR}/wendy_patched/nimble_host/include" IN_LIST actual)
            message(FATAL_ERROR "Missing copied NimBLE headers in ${property}: ${actual}")
        endif()
    endforeach()
    file(READ "${CMAKE_BINARY_DIR}/wendy_patched/nimble_host/src/ble_l2cap_sig.c" actual)
    if(NOT actual MATCHES "ble_l2cap_sig_extract_chan_disconnect" OR
       NOT actual MATCHES "ble_l2cap_sig_le_credits_locked")
        message(FATAL_ERROR "NimBLE signaling patches missing")
    endif()
    file(READ "${CMAKE_BINARY_DIR}/wendy_patched/nimble_host/src/ble_l2cap_coc.c" actual)
    if(NOT actual MATCHES "ble_l2cap_coc_credit_timer" OR
       NOT actual MATCHES "ble_l2cap_coc_credit_target")
        message(FATAL_ERROR "NimBLE credit patches missing")
    endif()
    file(READ "${CMAKE_BINARY_DIR}/wendy_patched/nimble_host/src/ble_hs_hci.c" actual)
    if(NOT actual MATCHES "ble_hs_sched_tx_retry")
        message(FATAL_ERROR "NimBLE ACL patch missing")
    endif()
endforeach()
