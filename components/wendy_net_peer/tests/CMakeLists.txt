function(format_test name source)
    add_executable(format_${name} ${source})
    target_link_libraries(format_${name} PRIVATE peer_format)
    target_compile_options(format_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME format_${name} COMMAND format_${name} ${ARGN})
endfunction()
