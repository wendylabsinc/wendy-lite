cmake_minimum_required(VERSION 3.20)
project(wendy_net_format_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
# Reuse the production crypto sources and pinned build-local dependency patches.
# Only format_* targets/tests are needed for this standalone component review.
add_subdirectory(../../wendy_pki/tests pki EXCLUDE_FROM_ALL)
set(CJSON_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../managed_components/espressif__cjson/cJSON" CACHE PATH "Pinned ESP cJSON source directory")
if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
    message(FATAL_ERROR "Resolve managed espressif/cjson with ESP-IDF, or set CJSON_ROOT to its cJSON source directory")
endif()
add_library(format_cjson STATIC "${CJSON_ROOT}/cJSON.c")
target_include_directories(format_cjson PUBLIC "${CJSON_ROOT}")
set(format_sources ../src/protocol.c ../src/binary.c ../src/packet.c ../src/directory.c ../src/manifest_service.c ../src/service_protocol.c ../src/guest_json.c ../src/certificate_fetch.c ../src/certificate_store.c ../src/certificate_wire.c)
add_library(peer_format STATIC ${format_sources})
target_include_directories(peer_format BEFORE PUBLIC "${CJSON_ROOT}")
target_include_directories(peer_format PUBLIC ../include PRIVATE ../src)
target_link_libraries(peer_format PUBLIC pki format_cjson)
target_compile_options(peer_format PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
function(format_test name source)
    add_executable(format_${name} ${source})
    target_link_libraries(format_${name} PRIVATE peer_format)
    target_compile_options(format_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME format_${name} COMMAND format_${name} ${ARGN})
endfunction()
format_test(device_dns device_dns_success_test.c)
format_test(directory directory_test.c "${CMAKE_CURRENT_LIST_DIR}/../../wendy_pki/tests/fixtures/device-key.der")
format_test(key_fifo directory_key_fifo_success_test.c "${CMAKE_CURRENT_LIST_DIR}/../../wendy_pki/tests/fixtures/device-key.der")
format_test(endpoint_cache endpoint_cache_success_test.c)
format_test(certificate_fetch certificate_fetch_success_test.c)
format_test(certificate_store certificate_store_success_test.c)
format_test(service_schema service_schema_success_test.c)
# This target compiles the real cache/fetch and ledger together, so outstanding
# immutable loans remain charged after cache close and runtime shutdown.
add_library(peer_format_budget STATIC ${format_sources} ../../wendy_net_common/src/runtime_memory.c)
target_include_directories(peer_format_budget BEFORE PUBLIC "${CJSON_ROOT}")
target_include_directories(peer_format_budget PUBLIC ../include PRIVATE ../src)
target_compile_definitions(peer_format_budget PUBLIC WENDY_NET_MEMORY_HOST_BUDGET_TEST=1)
target_compile_options(peer_format_budget PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
target_link_libraries(peer_format_budget PUBLIC pki format_cjson)
foreach(fixture certificate_fetch service_schema)
    if(fixture STREQUAL "certificate_fetch")
        set(source certificate_fetch_ram_success_test.c)
    else()
        set(source service_schema_success_test.c)
    endif()
    add_executable(format_${fixture}_budget ${source})
    target_link_libraries(format_${fixture}_budget PRIVATE peer_format_budget)
    target_compile_options(format_${fixture}_budget PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME format_${fixture}_budget COMMAND format_${fixture}_budget)
endforeach()

add_executable(format_manifest_interop manifest_interop_test.c)
target_link_libraries(format_manifest_interop PRIVATE peer_format)
target_compile_options(format_manifest_interop PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME format_manifest_interop COMMAND "${GO_EXECUTABLE}" run
    "${CMAKE_CURRENT_LIST_DIR}/manifest_interop_driver.go" "$<TARGET_FILE:format_manifest_interop>")
set_tests_properties(format_manifest_interop PROPERTIES TIMEOUT 60)
