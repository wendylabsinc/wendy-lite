cmake_minimum_required(VERSION 3.20)
project(native_settings_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
find_package(Threads REQUIRED)
foreach(profile full constrained no_ble no_wifi)
 add_executable(settings_${profile} settings_success_test.c)
 target_include_directories(settings_${profile} BEFORE PRIVATE . ../../../wendy_pki/tests/stubs)
 target_include_directories(settings_${profile} PRIVATE ../../src ../../../wendy_com/include ../../../wendy_conf/include ../../../nanopb ../../../wendy_net_common/include)
 target_compile_definitions(settings_${profile} PRIVATE CONFIG_WENDY_NET_MAX_LINKS=6 CONFIG_WENDY_NET_BLE_CHANNELS=3)
 if(profile STREQUAL constrained)
  target_compile_definitions(settings_${profile} PRIVATE CONFIG_WENDY_NET_CONSTRAINED_PROFILE=1)
 else()
  target_compile_definitions(settings_${profile} PRIVATE TEST_PSRAM=1)
 endif()
 if(NOT profile STREQUAL no_ble)
  target_compile_definitions(settings_${profile} PRIVATE CONFIG_WENDY_NET_BLE=1)
 endif()
 if(NOT profile STREQUAL no_wifi)
  target_compile_definitions(settings_${profile} PRIVATE CONFIG_WENDY_WIFI_ENABLED=1)
 endif()
 target_compile_options(settings_${profile} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
 target_link_libraries(settings_${profile} PRIVATE Threads::Threads)
 add_test(NAME settings_${profile}_success COMMAND settings_${profile})
endforeach()
