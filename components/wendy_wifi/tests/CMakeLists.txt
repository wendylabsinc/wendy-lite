cmake_minimum_required(VERSION 3.16)
project(wendy_wifi_services_test C)
set(CMAKE_C_STANDARD 11)
find_package(Threads REQUIRED)
enable_testing()
add_executable(services_test services_test.c)
target_include_directories(services_test PRIVATE stubs ../include)
target_link_libraries(services_test PRIVATE Threads::Threads)
target_compile_options(services_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME wifi_services_and_reconnection COMMAND services_test)
set_tests_properties(wifi_services_and_reconnection PROPERTIES TIMEOUT 10)
