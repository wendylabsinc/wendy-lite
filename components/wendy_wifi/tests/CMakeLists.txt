cmake_minimum_required(VERSION 3.16)
project(wendy_wifi_services_test C)
set(CMAKE_C_STANDARD 11)
find_package(Threads REQUIRED)
enable_testing()
add_executable(services_test services_test.c)
target_include_directories(services_test PRIVATE stubs ../include)
target_link_libraries(services_test PRIVATE Threads::Threads)
target_compile_options(services_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME wifi_services_and_reconnection COMMAND services_test)
set_tests_properties(wifi_services_and_reconnection PROPERTIES TIMEOUT 10)

add_executable(no_radio_success_test no_radio_success_test.c)
target_include_directories(no_radio_success_test PRIVATE stubs ../include)
target_compile_definitions(no_radio_success_test PRIVATE CONFIG_WENDY_NET_TEST_NO_WIFI_RADIO=1 CONFIG_WENDY_NET_PEER=1)
target_link_libraries(no_radio_success_test PRIVATE Threads::Threads)
target_compile_options(no_radio_success_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME wifi_no_radio_boot COMMAND no_radio_success_test)

add_executable(normal_init_success_test no_radio_success_test.c)
target_include_directories(normal_init_success_test PRIVATE stubs ../include)
target_link_libraries(normal_init_success_test PRIVATE Threads::Threads)
target_compile_options(normal_init_success_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME wifi_normal_init COMMAND normal_init_success_test)

add_executable(ble_only_radio_success_test no_radio_success_test.c)
target_include_directories(ble_only_radio_success_test PRIVATE stubs ../include)
target_compile_definitions(ble_only_radio_success_test PRIVATE CONFIG_WENDY_NET_BLE_ONLY_RADIO=1 CONFIG_WENDY_NET_PEER=1)
target_link_libraries(ble_only_radio_success_test PRIVATE Threads::Threads)
target_compile_options(ble_only_radio_success_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME wifi_ble_only_profile COMMAND ble_only_radio_success_test)

add_executable(camera_services_test services_test.c)
target_include_directories(camera_services_test PRIVATE stubs ../include)
target_link_libraries(camera_services_test PRIVATE Threads::Threads)
target_compile_options(camera_services_test PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_compile_definitions(camera_services_test PRIVATE CONFIG_WENDY_SENSOR_LINK_XIAO_ESP32S3=1)
add_test(NAME wifi_camera_services_and_reconnection COMMAND camera_services_test)
set_tests_properties(wifi_camera_services_and_reconnection PROPERTIES TIMEOUT 10)
