cmake_minimum_required(VERSION 3.16)
project(wendy_net_memory_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(wendy_net_memory_test test_memory.c ../src/memory.c)
target_include_directories(wendy_net_memory_test PRIVATE ../include)
target_link_libraries(wendy_net_memory_test PRIVATE Threads::Threads)
target_compile_options(wendy_net_memory_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME memory_accounting COMMAND wendy_net_memory_test)

add_executable(wendy_net_runtime_budget_test test_runtime_memory.c ../src/memory.c ../src/runtime_memory.c)
target_include_directories(wendy_net_runtime_budget_test PRIVATE ../include)
target_compile_definitions(wendy_net_runtime_budget_test PRIVATE WENDY_NET_MEMORY_HOST_BUDGET_TEST=1)
target_link_libraries(wendy_net_runtime_budget_test PRIVATE Threads::Threads)
target_compile_options(wendy_net_runtime_budget_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME runtime_memory_budget COMMAND wendy_net_runtime_budget_test)

add_executable(wendy_net_runtime_libc_test test_runtime_memory.c)
target_include_directories(wendy_net_runtime_libc_test PRIVATE ../include)
target_compile_options(wendy_net_runtime_libc_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME runtime_memory_host_libc COMMAND wendy_net_runtime_libc_test)

add_executable(wendy_net_binary_test test_binary.c ../src/binary.c)
target_include_directories(wendy_net_binary_test PRIVATE ../include)
target_compile_options(wendy_net_binary_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME binary_wire_writer COMMAND wendy_net_binary_test)

add_executable(wendy_net_task_test test_task.c ../src/task.c ../src/memory.c ../src/runtime_memory.c)
target_include_directories(wendy_net_task_test PRIVATE stubs ../include)
target_compile_definitions(wendy_net_task_test PRIVATE CONFIG_WENDY_NET_PEER=1 WENDY_NET_MEMORY_HOST_BUDGET_TEST=1)
target_link_libraries(wendy_net_task_test PRIVATE Threads::Threads)
target_compile_options(wendy_net_task_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME task_reservation_lifecycle COMMAND wendy_net_task_test)

add_executable(wendy_net_runtime_constrained_test test_runtime_memory.c ../src/memory.c ../src/runtime_memory.c)
target_include_directories(wendy_net_runtime_constrained_test PRIVATE ../include)
target_compile_definitions(wendy_net_runtime_constrained_test PRIVATE WENDY_NET_MEMORY_HOST_BUDGET_TEST=1 CONFIG_WENDY_NET_INTERNAL_MEMORY_LIMIT_KIB=160)
target_link_libraries(wendy_net_runtime_constrained_test PRIVATE Threads::Threads)
target_compile_options(wendy_net_runtime_constrained_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
add_test(NAME runtime_memory_constrained_budget COMMAND wendy_net_runtime_constrained_test)

foreach(profile IN ITEMS full constrained)
    add_executable(profile_${profile}_success_test profile_success_test.c)
    target_include_directories(profile_${profile}_success_test PRIVATE ../include)
    if(profile STREQUAL "full")
        target_compile_definitions(profile_${profile}_success_test PRIVATE CONFIG_WENDY_NET_CONSTRAINED_PROFILE=0 EXPECTED_NON_TRANSIT=0)
    else()
        target_compile_definitions(profile_${profile}_success_test PRIVATE CONFIG_WENDY_NET_CONSTRAINED_PROFILE=1 EXPECTED_NON_TRANSIT=1)
    endif()
    target_compile_options(profile_${profile}_success_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME profile_${profile}_policy COMMAND profile_${profile}_success_test)
endforeach()
