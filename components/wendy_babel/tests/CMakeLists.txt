cmake_minimum_required(VERSION 3.20)
project(wendy_babel_host C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(wendy_babel ../src/wire.c ../src/engine.c ../src/history.c ../src/storage.c)
target_include_directories(wendy_babel PUBLIC ../include PRIVATE ../../wendy_net_common/include)
target_compile_options(wendy_babel PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
add_executable(babel_test babel_test.c)
target_link_libraries(babel_test PRIVATE wendy_babel)
target_compile_options(babel_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME babel_core COMMAND babel_test)
add_executable(babel_driver driver.c)
target_link_libraries(babel_driver PRIVATE wendy_babel)

add_executable(babel_memory_test memory_test.c)
target_link_libraries(babel_memory_test PRIVATE wendy_babel)
target_compile_options(babel_memory_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME babel_sparse_memory COMMAND babel_memory_test)
add_test(NAME babel_steady_workspace_success COMMAND babel_memory_test --normal)
