if(CONFIG_WENDY_PKI)
    foreach(kind DEVICE TSA HTTPS)
        string(TOLOWER "${kind}" lower_kind)
        if("${CONFIG_WENDY_PKI_${kind}_ROOTS}" STREQUAL "")
            file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${lower_kind}_roots.pem" "")
            continue()
        endif()
        get_filename_component(root_bundle "${CONFIG_WENDY_PKI_${kind}_ROOTS}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
        if(NOT EXISTS "${root_bundle}")
            message(FATAL_ERROR "WENDY_PKI_${kind}_ROOTS must name a pre-pinned PEM bundle")
        endif()
        file(READ "${root_bundle}" bundle)
        if(NOT bundle MATCHES "-----BEGIN CERTIFICATE-----")
            message(FATAL_ERROR "WENDY_PKI_${kind}_ROOTS is not a PEM certificate bundle")
        endif()
        string(TOLOWER "${kind}" lower_kind)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${lower_kind}_roots.pem" "${bundle}")
    endforeach()
    idf_component_register(SRCS "src/wendy_pki_verify.c" "src/wendy_pki.c" "src/wendy_pki_tls.c"
        "src/wendy_roughtime.c" "src/wendy_roughtime_servers.c" "src/wendy_roughtime_network.c"
        INCLUDE_DIRS "include" REQUIRES wendy_wolfssl wendy_conf nvs_flash esp_timer esp_hw_support json http_parser lwip)
    foreach(kind device tsa https)
        target_add_binary_data(${COMPONENT_LIB} "${CMAKE_CURRENT_BINARY_DIR}/${kind}_roots.pem" TEXT RENAME_TO "wendy_pki_${kind}_roots")
    endforeach()
else()
    idf_component_register(SRCS "src/wendy_pki_stub.c" INCLUDE_DIRS "include" REQUIRES wendy_wolfssl wendy_conf nvs_flash esp_timer esp_hw_support json http_parser lwip)
endif()
