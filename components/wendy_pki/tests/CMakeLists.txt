cmake_minimum_required(VERSION 3.20)
project(wendy_pki_test C)
set(WOLF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../wendy_wolfssl/wolfssl")
file(GLOB wolf_src "${WOLF_ROOT}/src/*.c" "${WOLF_ROOT}/wolfcrypt/src/*.c")
list(FILTER wolf_src EXCLUDE REGEX "/(evp|bio|misc|ext_mlkem|ext_lms|ext_xmss)\\.c$")
add_library(wolf STATIC ${wolf_src} ../../wendy_wolfssl/random.c)
target_compile_definitions(wolf PUBLIC WOLFSSL_USER_SETTINGS)
target_include_directories(wolf PUBLIC "${WOLF_ROOT}" "${CMAKE_CURRENT_LIST_DIR}/../../wendy_wolfssl/include")
add_library(pki STATIC ../src/wendy_pki_verify.c)
target_include_directories(pki PUBLIC ../include)
target_link_libraries(pki PUBLIC wolf)
add_executable(verify_test verify_test.c)
target_link_libraries(verify_test PRIVATE pki)

enable_testing()
add_test(NAME pki_verification COMMAND verify_test "${CMAKE_CURRENT_LIST_DIR}/fixtures")
