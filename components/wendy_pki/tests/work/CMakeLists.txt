cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(wendy_pki_work_tests C)
    enable_testing()
    find_package(Threads REQUIRED)
    add_library(net_common STATIC ../../../wendy_net_common/src/memory.c ../../../wendy_net_common/src/runtime_memory.c)
    target_include_directories(net_common PUBLIC ../../../wendy_net_common/include)
    target_link_libraries(net_common PUBLIC Threads::Threads)
endif()
add_library(pki_work_core STATIC ../../src/wendy_pki_work.c)
target_include_directories(pki_work_core PUBLIC ../../include)
target_link_libraries(pki_work_core PUBLIC net_common Threads::Threads)
target_compile_options(pki_work_core PRIVATE -Wall -Wextra -Werror)
foreach(fixture success scope_success)
    add_executable(pki_work_${fixture}_test ../work_${fixture}_test.c)
    target_link_libraries(pki_work_${fixture}_test PRIVATE pki_work_core)
    target_compile_options(pki_work_${fixture}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME pki_work_${fixture} COMMAND pki_work_${fixture}_test)
    set_tests_properties(pki_work_${fixture} PROPERTIES TIMEOUT 20)
endforeach()
foreach(fixture dispatch lifecycle)
    add_executable(pki_work_${fixture}_test ../work_${fixture}_success_test.c)
    target_include_directories(pki_work_${fixture}_test PRIVATE ../work_dispatch_stubs ../../include ../../../wendy_net_common/include)
    target_compile_definitions(pki_work_${fixture}_test PRIVATE ESP_PLATFORM=1)
    target_compile_options(pki_work_${fixture}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(pki_work_${fixture}_test PRIVATE Threads::Threads)
    add_test(NAME pki_work_${fixture} COMMAND pki_work_${fixture}_test)
    set_tests_properties(pki_work_${fixture} PROPERTIES TIMEOUT 20)
endforeach()
