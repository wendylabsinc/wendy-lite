cmake_minimum_required(VERSION 3.16)
project(wendy_cloud_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
find_package(Threads REQUIRED)
add_executable(cloud_reconnect_success_test cloud_reconnect_success_test.c)
target_include_directories(cloud_reconnect_success_test PRIVATE stubs ../include ../../wendy_pki/include ../../wendy_conf/include ../../nanopb)
target_compile_definitions(cloud_reconnect_success_test PRIVATE CONFIG_WENDY_CLOUD=1 CONFIG_WENDY_CLOUD_TASK_STACK_SIZE=24576 CONFIG_WENDY_CLOUD_TASK_PRIORITY=5 CONFIG_WENDY_CLOUD_TASK_CORE_AFFINITY=0 CONFIG_WENDY_CLOUD_RECONNECT_DELAY_MS=5000)
target_compile_options(cloud_reconnect_success_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(cloud_reconnect_success_test PRIVATE Threads::Threads)
add_test(NAME cloud_reconnect_success COMMAND cloud_reconnect_success_test)
set_tests_properties(cloud_reconnect_success PROPERTIES TIMEOUT 5)
