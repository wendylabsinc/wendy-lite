cmake_minimum_required(VERSION 3.20)
project(wendy_net_raw_success C)
enable_testing()
find_package(Threads REQUIRED)
add_executable(raw_socket_success_test raw_socket_success_test.c ../src/raw.c)
target_include_directories(raw_socket_success_test PRIVATE ../include)
target_compile_features(raw_socket_success_test PRIVATE c_std_11)
target_compile_options(raw_socket_success_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(raw_socket_success_test PRIVATE Threads::Threads)
add_test(NAME raw_socket_success COMMAND raw_socket_success_test)
set_tests_properties(raw_socket_success PROPERTIES TIMEOUT 10)
